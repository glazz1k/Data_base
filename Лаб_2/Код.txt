IF OBJECT_ID('Sale', 'U') IS NOT NULL DROP TABLE Sale;
IF OBJECT_ID('Edition', 'U') IS NOT NULL DROP TABLE Edition;
IF OBJECT_ID('Book_Author', 'U') IS NOT NULL DROP TABLE Book_Author;
IF OBJECT_ID('Book_Genre', 'U') IS NOT NULL DROP TABLE Book_Genre;
IF OBJECT_ID('Book', 'U') IS NOT NULL DROP TABLE Book;
IF OBJECT_ID('Author', 'U') IS NOT NULL DROP TABLE Author;
IF OBJECT_ID('Publisher', 'U') IS NOT NULL DROP TABLE Publisher;
IF OBJECT_ID('Genre', 'U') IS NOT NULL DROP TABLE Genre;

CREATE TABLE Book
(
	code NVARCHAR(50) PRIMARY KEY,
	title NVARCHAR(100) NOT NULL,
	year_published INT NOT NULL,
	[language] NVARCHAR(50) NOT NULL
)

CREATE TABLE Author
(
	id INT PRIMARY KEY IDENTITY(1,1),
	full_name NVARCHAR(200) NOT NULL,
	birthday_date DATE NOT NULL,
	country NVARCHAR(50) NOT NULL
)

CREATE TABLE Publisher
(
	id INT PRIMARY KEY IDENTITY(1,1),
	[name] NVARCHAR(100) NOT NULL UNIQUE,
	city NVARCHAR(50) NOT NULL,
	[address] NVARCHAR(200) NOT NULL,
	email NVARCHAR(200) NOT NULL UNIQUE
)

CREATE TABLE Edition
(
	id INT PRIMARY KEY IDENTITY(1,1),
	purchase_price MONEY NOT NULL CHECK (purchase_price > 0),
	[count] INT NOT NULL CHECK ([count] > 0),
	arrival_date DATE NOT NULL,
	book_code NVARCHAR(50) NOT NULL,
	publisher_id INT NOT NULL,
	FOREIGN KEY (book_code) REFERENCES Book(code) ON DELETE CASCADE,
	FOREIGN KEY (publisher_id) REFERENCES Publisher(id) ON DELETE NO ACTION
)

CREATE TABLE Sale
(
	id INT PRIMARY KEY IDENTITY(1,1),
	price MONEY NOT NULL CHECK (price > 0),
	[count] INT NOT NULL CHECK ([count] > 0),
	[date] DATE NOT NULL,
	discount DECIMAL(5,2) NOT NULL CHECK (discount >= 0 AND discount <= 100) DEFAULT 0,
	edition_id INT NOT NULL,
	FOREIGN KEY (edition_id) REFERENCES Edition(id) ON DELETE NO ACTION
)

CREATE TABLE Genre
(
	id INT PRIMARY KEY IDENTITY(1,1),
	[name] NVARCHAR(50) NOT NULL UNIQUE,
	[description] NVARCHAR(1000)
)

CREATE TABLE Book_Author
(
	book_code NVARCHAR(50) NOT NULL,
	author_id INT NOT NULL,
	FOREIGN KEY (book_code) REFERENCES Book(code) ON DELETE CASCADE,
	FOREIGN KEY (author_id) REFERENCES Author(id) ON DELETE CASCADE,
	PRIMARY KEY(book_code, author_id)
)

CREATE TABLE Book_Genre
(
	genre_id INT NOT NULL,
	book_code NVARCHAR(50) NOT NULL,
	FOREIGN KEY (genre_id) REFERENCES Genre(id) ON DELETE CASCADE,
	FOREIGN KEY (book_code) REFERENCES Book(code) ON DELETE CASCADE,
	PRIMARY KEY(genre_id, book_code)
)
